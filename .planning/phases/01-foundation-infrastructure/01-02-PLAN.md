```python
from app.workers.celery_app import celery_app
from app.workers.tasks.test_task import test_echo_task
from app.schemas.tasks import TaskDispatchRequest, TaskDispatchResponse, TaskResultResponse

tasks_router = APIRouter(prefix="/api/v1/tasks", tags=["tasks"])

@tasks_router.post("/dispatch", response_model=TaskDispatchResponse)
async def dispatch_task(req: TaskDispatchRequest):
    result = test_echo_task.delay(req.message)
    return TaskDispatchResponse(task_id=result.id, status="queued")

@tasks_router.get("/{task_id}", response_model=TaskResultResponse)
async def get_task_result(task_id: str):
    result = AsyncResult(task_id, app=celery_app)
    if result.state == "PENDING":
        raise HTTPException(status_code=404, detail="Task not found")
    return TaskResultResponse(
        task_id=task_id,
        status=result.state,
        result=result.result if result.ready() else None,
    )
```

**1. Create app/services/oss.py** — boto3 S3 wrapper (per D-06, D-07)

```python
import boto3
from botocore.exceptions import ClientError
from app.core.config import settings
import uuid

def get_s3_client():
    """Create boto3 S3 client (per D-07: all config from env vars)."""
    return boto3.client(
        "s3",
        endpoint_url=settings.S3_ENDPOINT_URL,
        aws_access_key_id=settings.S3_ACCESS_KEY,
        aws_secret_access_key=settings.S3_SECRET_KEY,
        region_name=settings.S3_REGION,
    )

def ensure_bucket(s3_client=None, bucket: str = None):
    """Create bucket if not exists. Called on startup."""
    client = s3_client or get_s3_client()
    bucket = bucket or settings.S3_BUCKET
    try:
        client.head_bucket(Bucket=bucket)
    except ClientError:
        client.create_bucket(Bucket=bucket)

def generate_upload_url(extension: str = "jpg", prefix: str = "uploads") -> dict:
    """Generate presigned PUT URL (300s TTL per Pitfall #7)."""
    client = get_s3_client()
    key = f"{prefix}/{uuid.uuid4().hex}.{extension}"
    url = client.generate_presigned_url(
        "put_object",
        Params={"Bucket": settings.S3_BUCKET, "Key": key},
        ExpiresIn=300,
    )
    return {"upload_url": url, "key": key}

def generate_download_url(key: str) -> dict:
    """Generate presigned GET URL (3600s TTL per Pitfall #15)."""
    client = get_s3_client()
    url = client.generate_presigned_url(
        "get_object",
        Params={"Bucket": settings.S3_BUCKET, "Key": key},
        ExpiresIn=3600,
    )
    return {"download_url": url, "key": key}

def object_exists(key: str) -> bool:
    """Check if object exists in bucket."""
    client = get_s3_client()
    try:
        client.head_object(Bucket=settings.S3_BUCKET, Key=key)
        return True
    except ClientError:
        return False
```

```python
from fastapi import APIRouter, HTTPException
from app.schemas.oss import (
    PresignUploadRequest, PresignUploadResponse,
    PresignDownloadRequest, PresignDownloadResponse,
    ConfirmUploadRequest, ConfirmUploadResponse,
)
from app.services.oss import generate_upload_url, generate_download_url, object_exists

oss_router = APIRouter(prefix="/api/v1/oss", tags=["oss"])

@oss_router.post("/presign-upload", response_model=PresignUploadResponse)
async def presign_upload(req: PresignUploadRequest):
    result = generate_upload_url(extension=req.extension)
    return PresignUploadResponse(**result)

@oss_router.post("/presign-download", response_model=PresignDownloadResponse)
async def presign_download(req: PresignDownloadRequest):
    if not object_exists(req.key):
        raise HTTPException(status_code=404, detail="Object not found")
    result = generate_download_url(key=req.key)
    return PresignDownloadResponse(**result)

@oss_router.post("/confirm-upload", response_model=ConfirmUploadResponse)
async def confirm_upload(req: ConfirmUploadRequest):
    exists = object_exists(req.key)
    return ConfirmUploadResponse(key=req.key, exists=exists)
```
